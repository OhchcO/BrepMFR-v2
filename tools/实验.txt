# 使用说明
# 1. 每次只复制一个实验标题下面的完整代码块到 PowerShell。
# 2. 每组固定使用 1% 数据、随机种子 2026、100 轮、BatchSize 16。
# 3. 训练结束后会自动寻找该实验最新的 best-*.ckpt，然后运行完整 test.txt。
# 4. 所有实验共用已有 BIN 缓存，不要添加 -RebuildPreparedData。
# 5. oracle 使用 GT 实例信息，只表示理论性能上界，不能作为正常推理结果。
# 6. 实验 11-14 是正向映射（positive_only）消融，用于锁定论文对照链：
#    Baseline → dynamic_no_loss+positive → affinity_only+positive
#    → instance+positive → oracle+positive


# ============================================================
# 1. baseline：纯基线（已完成，可按需重新运行）
# ============================================================

powershell.exe -ExecutionPolicy Bypass -File `
  "D:\Code\BrepMFR\tools\run_mfrinstseg_experiments.ps1" `
  -Mode baseline `
  -RawDatasetPath "D:\Code\Datasets\MFRInstSeg" `
  -PreparedDatasetPath "D:\Code\Datasets\MFRInstSeg_BrepMFR_SPEPCAAG" `
  -SamplePercent 0.01 `
  -Seed 2026 `
  -BatchSize 16 `
  -MaxEpochs 100 `
  -NumWorkers 0

$ResultRoot = "D:\Code\BrepMFR\results\MFRInstSeg_1pct_baseline_seed2026"
$Checkpoint = Get-ChildItem -LiteralPath $ResultRoot -Recurse -Filter "best-*.ckpt" `
  | Sort-Object LastWriteTime -Descending `
  | Select-Object -First 1 -ExpandProperty FullName
if (-not $Checkpoint) {
  throw "No best checkpoint found for baseline"
}
Write-Host "Testing checkpoint: $Checkpoint"
powershell.exe -ExecutionPolicy Bypass -File `
  "D:\Code\BrepMFR\tools\run_mfrinstseg_test.ps1" `
  -Checkpoint $Checkpoint `
  -PreparedDataset "D:\Code\Datasets\MFRInstSeg_BrepMFR_SPEPCAAG" `
  -BatchSize 16 `
  -NumWorkers 0
